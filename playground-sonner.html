Copy-paste usage

<link rel="stylesheet" href="assets/css/sonner.css">
<script src="assets/js/sonner.js"></script>
<script>
  // auto-mounted Toaster (position/theme/dir follow settings)
  toast.success('Saved', { description: 'Hello world', richColors: true });
  toast('With action', { action: { label: 'Undo', onClick: () => {} } });
  toast.promise(fetch('/api').then(r => r.json()), {
    loading: 'Loading…', success: (d) => 'Done', error: 'Failed'
  });
  const id = toast.loading('Uploading…');
  toast.success('Uploaded', { id });   // update in place by id
  toast.success('Colored', { richColors: true, description: 'full-color variant' });
  toast('Moved', { position: 'top-center' });  // per-toast dock override
  toast('Gone soon', { onAutoClose: (t) => console.log('auto', t.id) });
  toast.dismiss();            // dismiss all
  toast.dismiss(id);          // dismiss one
  Sonner.setPosition('top-center');
  Sonner.createToaster({ position: 'top-center', richColors: true,
    visibleToasts: 5, expand: true, gap: 14 });
</script>
